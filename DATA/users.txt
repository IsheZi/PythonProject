import bcrypt
from data.db import connect_db

def migrate_users(file_path="users.txt"):
    conn = connect_db()
    cursor = conn.cursor()

    with open(file_path, "r") as f:
        for line in f:
            parts = line.strip().split(",")
            if len(parts) >= 3:
                username, hashed_pw, role = parts
                cursor.execute("""INSERT OR IGNORE INTO users (username, password_hash, role)
                                  VALUES (?, ?, ?)""", (username, hashed_pw, role))
    conn.commit()
    conn.close()
